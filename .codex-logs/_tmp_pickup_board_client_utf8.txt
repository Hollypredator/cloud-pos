"use client";

import { useEffect, useState } from "react";
import type { Order } from "@/lib/types";

export function PickupBoardClient({
  initialPreparing,
  initialReady,
}: {
  initialPreparing: Order[];
  initialReady: Order[];
}) {
  const [preparing, setPreparing] = useState(initialPreparing);
  const [ready, setReady] = useState(initialReady);

  // Simple polling for now, since we don't have a shared websocket/pusher setup 
  // that I can easily tap into without knowing the config.
  useEffect(() => {
    const interval = setInterval(async () => {
      try {
        const res = await fetch("/api/ops/pickup-snapshot");
        if (res.ok) {
          const data = await res.json();
          setPreparing(data.preparing);
          setReady(data.ready);
        }
      } catch (e) {
        console.error("Failed to poll pickup snapshot", e);
      }
    }, 5000);
    return () => clearInterval(interval);
  }, []);

  return (
    <div className="grid grid-cols-2 h-screen">
      {/* Preparing Column */}
      <div className="bg-[#111114] border-r border-white/5 p-12">
        <div className="flex items-center justify-between mb-12">
          <h2 className="text-4xl font-bold text-slate-400 tracking-tight">Hazırlanıyor</h2>
          <div className="px-4 py-1 rounded-full bg-sky-500/10 text-sky-400 text-sm font-bold border border-sky-500/20 uppercase tracking-widest">
            Preparing
          </div>
        </div>
        
        <div className="grid grid-cols-2 gap-6">
          {preparing.map((order) => (
            <div key={order.id} className="bg-white/5 rounded-3xl p-8 border border-white/5 animate-pulse">
              <div className="text-5xl font-black text-white mb-2">
                #{order.check_number || order.id.slice(0, 4)}
              </div>
              <div className="text-xl text-slate-400 font-medium truncate">
                {order.customer_name || "Misafir"}
              </div>
            </div>
          ))}
          {preparing.length === 0 && (
            <div className="col-span-2 text-slate-600 text-2xl font-medium mt-12">
              Henüz yeni sipariş yok...
            </div>
          )}
        </div>
      </div>

      {/* Ready Column */}
      <div className="bg-[#0a0a0c] p-12">
        <div className="flex items-center justify-between mb-12">
          <h2 className="text-4xl font-bold text-emerald-400 tracking-tight">Afiyet Olsun!</h2>
          <div className="px-4 py-1 rounded-full bg-emerald-500/10 text-emerald-400 text-sm font-bold border border-emerald-500/20 uppercase tracking-widest">
            Ready
          </div>
        </div>

        <div className="grid grid-cols-2 gap-8">
          {ready.map((order) => (
            <div key={order.id} className="bg-[linear-gradient(135deg,rgba(16,185,129,0.1)_0%,rgba(5,150,105,0.05)_100%)] rounded-[2.5rem] p-10 border border-emerald-500/20 shadow-[0_20px_50px_rgba(16,185,129,0.15)] relative overflow-hidden group">
              <div className="absolute top-0 right-0 p-4">
                 <div className="w-4 h-4 rounded-full bg-emerald-400 animate-ping" />
              </div>
              <div className="text-7xl font-black text-emerald-400 mb-4 drop-shadow-[0_0_15px_rgba(16,185,129,0.3)]">
                #{order.check_number || order.id.slice(0, 4)}
              </div>
              <div className="text-3xl text-white font-bold truncate">
                {order.customer_name || "Misafir"}
              </div>
            </div>
          ))}
          {ready.length === 0 && (
            <div className="col-span-2 text-slate-700 text-2xl font-medium mt-12">
              Bekleyen hazır sipariş yok.
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
